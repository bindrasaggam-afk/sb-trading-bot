package com.sb

import org.json.JSONObject
import java.io.IOException
import java.net.HttpURLConnection
import java.net.URL
import java.util.concurrent.ExecutorService
import java.util.concurrent.Executors

/**
 * SB BOTE Paper Trading API client.
 *
 * Uses the existing FastAPI backend.
 * No mock balances or mock orders are generated.
 */
object PaperTradingApi {

    // Replace with the reachable address of your EXISTING backend.
    // Android emulator + backend on development PC:
    // http://10.0.2.2:8000
    //
    // Physical Android phone + backend on PC:
    // http://YOUR_PC_LAN_IP:8000
    //
    // Cloud deployment:
    // Use your existing HTTPS backend URL.
    @Volatile
    var baseUrl: String = "http://10.0.2.2:8000"

    private const val CONNECT_TIMEOUT_MS = 8000
    private const val READ_TIMEOUT_MS = 10000

    private val executor: ExecutorService =
        Executors.newFixedThreadPool(2)

    data class ApiResult(
        val success: Boolean,
        val httpCode: Int? = null,
        val data: JSONObject? = null,
        val error: String? = null
    )

    /**
     * Execute a GET request away from the Android main thread.
     * Callback is delivered on the worker thread.
     */
    fun get(
        path: String,
        callback: (ApiResult) -> Unit
    ) {
        executor.execute {
            callback(request(path))
        }
    }

    private fun request(path: String): ApiResult {
        var connection: HttpURLConnection? = null

        return try {
            val root = baseUrl.trim().trimEnd('/')
            require(root.startsWith("http://") ||
                    root.startsWith("https://")) {
                "Backend URL must start with http:// or https://"
            }

            require(path.startsWith("/") && !path.startsWith("//")) {
                "Invalid API path"
            }

            connection = (URL(root + path)
                .openConnection() as HttpURLConnection).apply {
                requestMethod = "GET"
                connectTimeout = CONNECT_TIMEOUT_MS
                readTimeout = READ_TIMEOUT_MS
                setRequestProperty("Accept", "application/json")
            }

            val code = connection.responseCode
            val stream = if (code in 200..299) {
                connection.inputStream
            } else {
                connection.errorStream
            }

            val body = stream?.bufferedReader()?.use {
                it.readText()
            }.orEmpty()

            if (code !in 200..299) {
                ApiResult(
                    success = false,
                    httpCode = code,
                    error = "API returned HTTP $code" +
                        if (body.isNotBlank()) ": $body" else ""
                )
            } else {
                val json = JSONObject(body)
                ApiResult(
                    success = true,
                    httpCode = code,
                    data = json
                )
            }
        } catch (e: Exception) {
            ApiResult(
                success = false,
                error = when (e) {
                    is IOException ->
                        "Backend connection failed: ${e.message}"
                    else -> e.message ?: "Unknown API error"
                }
            )
        } finally {
            connection?.disconnect()
        }
    }

    /**
     * Health check: this endpoint was previously used by SB BOTE.
     */
    fun checkHealth(callback: (ApiResult) -> Unit) {
        get("/api/health", callback)
    }

    /**
     * These Paper Trading routes must be checked against
     * the current FastAPI implementation before using them.
     */
    fun getPaperStatus(callback: (ApiResult) -> Unit) {
        get("/api/paper/status", callback)
    }

    fun getPaperPositions(callback: (ApiResult) -> Unit) {
        get("/api/paper/positions", callback)
    }

    fun getPaperPnl(callback: (ApiResult) -> Unit) {
        get("/api/paper/pnl", callback)
    }
}

package com.sb.backend.database

import android.database.sqlite.SQLiteDatabase

/**
 * SB BOT - Backend Database Migration
 *
 * Existing DashboardDbHelper database version 4 से
 * version 5 पर migrate करने के लिए।
 *
 * यह migration existing tables या data को delete नहीं करती।
 */
object BackendMigration {

    /**
     * Existing main database का नया version.
     *
     * DashboardDbHelper अभी version 4 पर है।
     * Backend tables जोड़ने के बाद इसे version 5 करना है।
     */
    const val DATABASE_VERSION = 5

    /**
     * Fresh installation पर backend tables बनाना।
     *
     * इसे DashboardDbHelper.onCreate() से call करेंगे।
     */
    fun onCreate(db: SQLiteDatabase) {

        BackendSchema.create(db)
    }

    /**
     * Existing database upgrade.
     *
     * SQLiteOpenHelper स्वयं onUpgrade() को transaction
     * के अंदर चलाता है। इसलिए यहाँ अलग transaction शुरू
     * नहीं किया गया है।
     */
    fun onUpgrade(
        db: SQLiteDatabase,
        oldVersion: Int,
        newVersion: Int
    ) {

        if (oldVersion < 5 && newVersion >= 5) {

            BackendSchema.create(db)
        }
    }

    /**
     * Backend schema को सुरक्षित तरीके से उपलब्ध कराना।
     *
     * CREATE IF NOT EXISTS होने के कारण पहले से बनी
     * backend tables दोबारा create नहीं होंगी।
     */
    fun ensureCurrent(db: SQLiteDatabase) {

        BackendSchema.create(db)
    }
}

package com.sb.backend.database

import android.database.sqlite.SQLiteDatabase

/**
 * SB BOT - Backend Database Schema
 *
 * यह class केवल backend की नई tables और indexes बनाती है।
 * Existing DashboardDbHelper की tables को modify या delete नहीं करती।
 */
object BackendSchema {

    const val VERSION = 1

    const val TABLE_INSTRUMENTS = "backend_instruments"
    const val TABLE_QUOTES = "backend_quotes"
    const val TABLE_CANDLES = "backend_candles"
    const val TABLE_SIGNALS = "backend_signals"
    const val TABLE_METADATA = "backend_metadata"

    /**
     * सभी backend tables और indexes बनाता है।
     *
     * CREATE TABLE IF NOT EXISTS के कारण
     * existing data overwrite नहीं होगा।
     */
    fun create(db: SQLiteDatabase) {

        createMetadataTable(db)

        createInstrumentsTable(db)

        createQuotesTable(db)

        createCandlesTable(db)

        createSignalsTable(db)

        createIndexes(db)

        saveSchemaVersion(db)
    }

    // --------------------------------------------------
    // METADATA
    // --------------------------------------------------

    private fun createMetadataTable(db: SQLiteDatabase) {

        db.execSQL(
            """
            CREATE TABLE IF NOT EXISTS $TABLE_METADATA (
                meta_key TEXT PRIMARY KEY,
                meta_value TEXT NOT NULL,
                updated_at INTEGER NOT NULL
            )
            """.trimIndent()
        )
    }

    private fun saveSchemaVersion(db: SQLiteDatabase) {

        val now = System.currentTimeMillis()

        db.execSQL(
            """
            INSERT OR REPLACE INTO $TABLE_METADATA
            (
                meta_key,
                meta_value,
                updated_at
            )
            VALUES
            (
                'backend_schema_version',
                '$VERSION',
                $now
            )
            """.trimIndent()
        )
    }

    // --------------------------------------------------
    // INSTRUMENTS
    // --------------------------------------------------

    private fun createInstrumentsTable(db: SQLiteDatabase) {

        db.execSQL(
            """
            CREATE TABLE IF NOT EXISTS $TABLE_INSTRUMENTS (

                id INTEGER PRIMARY KEY AUTOINCREMENT,

                symbol TEXT NOT NULL,

                name TEXT NOT NULL,

                exchange TEXT NOT NULL,

                segment TEXT NOT NULL,

                instrument_type TEXT NOT NULL,

                trading_symbol TEXT NOT NULL,

                token TEXT,

                lot_size INTEGER NOT NULL DEFAULT 1,

                tick_size REAL NOT NULL DEFAULT 0.05,

                is_active INTEGER NOT NULL DEFAULT 1,

                created_at INTEGER NOT NULL,

                updated_at INTEGER NOT NULL,

                CHECK (lot_size > 0),

                CHECK (tick_size > 0),

                UNIQUE (
                    exchange,
                    trading_symbol
                )
            )
            """.trimIndent()
        )
    }

    // --------------------------------------------------
    // LIVE QUOTES
    // --------------------------------------------------

    private fun createQuotesTable(db: SQLiteDatabase) {

        db.execSQL(
            """
            CREATE TABLE IF NOT EXISTS $TABLE_QUOTES (

                id INTEGER PRIMARY KEY AUTOINCREMENT,

                symbol TEXT NOT NULL,

                exchange TEXT NOT NULL,

                ltp REAL NOT NULL,

                open_price REAL,

                high_price REAL,

                low_price REAL,

                previous_close REAL,

                volume INTEGER NOT NULL DEFAULT 0,

                change_value REAL,

                change_percent REAL,

                timestamp INTEGER NOT NULL,

                is_live INTEGER NOT NULL DEFAULT 0,

                CHECK (ltp >= 0),

                CHECK (volume >= 0),

                UNIQUE (
                    symbol,
                    exchange
                )
            )
            """.trimIndent()
        )
    }

    // --------------------------------------------------
    // HISTORICAL CANDLES
    // --------------------------------------------------

    private fun createCandlesTable(db: SQLiteDatabase) {

        db.execSQL(
            """
            CREATE TABLE IF NOT EXISTS $TABLE_CANDLES (

                id INTEGER PRIMARY KEY AUTOINCREMENT,

                symbol TEXT NOT NULL,

                exchange TEXT NOT NULL,

                timeframe TEXT NOT NULL,

                candle_timestamp INTEGER NOT NULL,

                open_price REAL NOT NULL,

                high_price REAL NOT NULL,

                low_price REAL NOT NULL,

                close_price REAL NOT NULL,

                volume INTEGER NOT NULL DEFAULT 0,

                is_complete INTEGER NOT NULL DEFAULT 1,

                CHECK (open_price >= 0),

                CHECK (high_price >= 0),

                CHECK (low_price >= 0),

                CHECK (close_price >= 0),

                CHECK (high_price >= low_price),

                CHECK (volume >= 0),

                UNIQUE (
                    symbol,
                    exchange,
                    timeframe,
                    candle_timestamp
                )
            )
            """.trimIndent()
        )
    }

    // --------------------------------------------------
    // TRADING SIGNALS
    // --------------------------------------------------

    private fun createSignalsTable(db: SQLiteDatabase) {

        db.execSQL(
            """
            CREATE TABLE IF NOT EXISTS $TABLE_SIGNALS (

                id INTEGER PRIMARY KEY AUTOINCREMENT,

                symbol TEXT NOT NULL,

                exchange TEXT NOT NULL,

                trading_style TEXT NOT NULL,

                signal_type TEXT NOT NULL,

                entry_price REAL,

                stop_loss REAL,

                target1 REAL,

                target2 REAL,

                target3 REAL,

                confidence REAL NOT NULL DEFAULT 0,

                timeframe TEXT NOT NULL DEFAULT '5m',

                status TEXT NOT NULL DEFAULT 'ACTIVE',

                reason TEXT NOT NULL DEFAULT '',

                created_at INTEGER NOT NULL,

                expires_at INTEGER,

                confirmation_count INTEGER NOT NULL DEFAULT 0,

                confirmation_required INTEGER NOT NULL DEFAULT 2,

                is_stable INTEGER NOT NULL DEFAULT 0,

                CHECK (
                    confidence >= 0
                    AND confidence <= 100
                ),

                CHECK (confirmation_count >= 0),

                CHECK (confirmation_required > 0)
            )
            """.trimIndent()
        )
    }

    // --------------------------------------------------
    // INDEXES
    // --------------------------------------------------

    private fun createIndexes(db: SQLiteDatabase) {

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_instruments_symbol
            ON $TABLE_INSTRUMENTS(symbol)
            """.trimIndent()
        )

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_instruments_exchange
            ON $TABLE_INSTRUMENTS(exchange)
            """.trimIndent()
        )

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_quotes_timestamp
            ON $TABLE_QUOTES(timestamp)
            """.trimIndent()
        )

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_candles_lookup
            ON $TABLE_CANDLES(
                symbol,
                exchange,
                timeframe,
                candle_timestamp
            )
            """.trimIndent()
        )

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_signals_symbol
            ON $TABLE_SIGNALS(symbol)
            """.trimIndent()
        )

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_signals_status
            ON $TABLE_SIGNALS(status)
            """.trimIndent()
        )

        db.execSQL(
            """
            CREATE INDEX IF NOT EXISTS
            idx_backend_signals_created
            ON $TABLE_SIGNALS(created_at)
            """.trimIndent()
        )
    }
}

package com.sb.backend.services

import android.content.Context

import com.sb.backend.repositories.SignalRepository
import com.sb.backend.repositories.PaperRepository

class BackendManager(context: Context) {

    private val appContext = context.applicationContext

    val database = BackendDatabase(appContext)
    private val dbHelper = database.databaseHelper

    // Repositories
    val instrumentRepository = InstrumentRepository(dbHelper)
    val marketRepository = MarketRepository(dbHelper)
    val signalRepository = SignalRepository(dbHelper)
    val paperRepository = PaperRepository(dbHelper)

    // Market Data
    val marketDataService = MarketDataService(
        marketRepository,
        instrumentRepository
    )

    // Indicators
    val indicatorService = IndicatorService

    // Signal Engine
    val signalEngine = SignalEngine(
        marketRepository = marketRepository,
        signalRepository = signalRepository
    )

    // Risk Manager
    val riskManager = RiskManager()

    // Paper Trading Engine
    val paperTradingEngine = PaperTradingEngine(
        paperRepository = paperRepository,
        marketRepository = marketRepository,
        riskManager = riskManager
    )

    @Volatile
    private var initialized = false

    @Synchronized
    fun initialize(): Boolean {
        return try {
            if (!database.isDatabaseOpen()) return false

            initialized = true
            true
        } catch (e: Exception) {
            initialized = false
            false
        }
    }

    fun isInitialized(): Boolean {
        return initialized && database.isDatabaseOpen()
    }

    fun getDatabaseStatus(): String {
        return if (isInitialized()) {
            "BACKEND_DATABASE_READY"
        } else {
            "BACKEND_DATABASE_NOT_READY"
        }
    }

    fun close() {
        initialized = false
        database.close()
    }
}

package com.sb.backend.trading

import com.sb.backend.core.AppConfig
import com.sb.backend.models.OrderSide
import com.sb.backend.models.Position
import kotlin.math.abs
import kotlin.math.floor

data class RiskCheckResult(
    val allowed: Boolean,
    val message: String,
    val requiredMargin: Double = 0.0,
    val availableCapital: Double = 0.0,
    val estimatedRisk: Double = 0.0
)

data class PositionSizingResult(
    val allowed: Boolean,
    val quantity: Int,
    val requiredMargin: Double,
    val estimatedRisk: Double,
    val message: String
)

object RiskManager {

    private fun isValidNumber(value: Double): Boolean {
        return !value.isNaN() && !value.isInfinite()
    }

    private fun normalizeSide(side: String): String {
        return side.trim().toUpperCase()
    }

    private fun marginRate(side: String): Double {
        return if (normalizeSide(side) == OrderSide.SELL.name) {
            AppConfig.get().shortMarginRate
        } else {
            1.0
        }
    }

    fun calculateRequiredMargin(
        side: String,
        quantity: Int,
        price: Double
    ): Double {
        if (quantity <= 0 || price <= 0.0 || !isValidNumber(price)) {
            return 0.0
        }

        val rate = marginRate(side)

        if (!isValidNumber(rate) || rate <= 0.0) {
            return 0.0
        }

        return quantity.toDouble() * price * rate
    }

    fun calculatePositionSize(
        capital: Double,
        riskPercent: Double,
        entryPrice: Double,
        stopLoss: Double,
        side: String
    ): PositionSizingResult {

        if (!isValidNumber(capital) || capital <= 0.0) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0, "Invalid capital"
            )
        }

        if (!isValidNumber(riskPercent) ||
            riskPercent <= 0.0 || riskPercent > 100.0
        ) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "Risk percent must be between 0 and 100"
            )
        }

        if (!isValidNumber(entryPrice) ||
            !isValidNumber(stopLoss) ||
            entryPrice <= 0.0 || stopLoss <= 0.0
        ) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "Invalid entry price or stop loss"
            )
        }

        val normalizedSide = normalizeSide(side)

        if (normalizedSide != OrderSide.BUY.name &&
            normalizedSide != OrderSide.SELL.name
        ) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0, "Invalid order side"
            )
        }

        val riskPerUnit = abs(entryPrice - stopLoss)

        if (riskPerUnit <= 0.0) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "Entry and stop loss cannot be equal"
            )
        }

        if (normalizedSide == OrderSide.BUY.name &&
            stopLoss >= entryPrice
        ) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "BUY stop loss must be below entry price"
            )
        }

        if (normalizedSide == OrderSide.SELL.name &&
            stopLoss <= entryPrice
        ) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "SELL stop loss must be above entry price"
            )
        }

        val allowedRisk = capital * riskPercent / 100.0
        val riskQuantity = floor(allowedRisk / riskPerUnit).toInt()

        if (riskQuantity <= 0) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "Capital is insufficient for this risk"
            )
        }

        val rate = marginRate(normalizedSide)

        if (!isValidNumber(rate) || rate <= 0.0) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0, "Invalid margin rate"
            )
        }

        val marginQuantity = floor(
            capital / (entryPrice * rate)
        ).toInt()

        val finalQuantity = minOf(riskQuantity, marginQuantity)

        if (finalQuantity <= 0) {
            return PositionSizingResult(
                false, 0, 0.0, 0.0,
                "Insufficient available capital"
            )
        }

        val requiredMargin = calculateRequiredMargin(
            normalizedSide, finalQuantity, entryPrice
        )

        val estimatedRisk = finalQuantity * riskPerUnit

        return PositionSizingResult(
            allowed = true,
            quantity = finalQuantity,
            requiredMargin = requiredMargin,
            estimatedRisk = estimatedRisk,
            message = "Position size calculated successfully"
        )
    }

    fun assessOrder(
        capital: Double,
        symbol: String,
        side: String,
        quantity: Int,
        entryPrice: Double,
        stopLoss: Double,
        openPositions: List<Position>
    ): RiskCheckResult {

        if (symbol.trim().isEmpty()) {
            return RiskCheckResult(false, "Symbol cannot be empty")
        }

        if (!isValidNumber(capital) || capital <= 0.0) {
            return RiskCheckResult(false, "Invalid capital")
        }

        if (quantity <= 0) {
            return RiskCheckResult(false, "Quantity must be greater than zero")
        }

        if (!isValidNumber(entryPrice) || entryPrice <= 0.0) {
            return RiskCheckResult(false, "Invalid entry price")
        }

        if (!isValidNumber(stopLoss) || stopLoss <= 0.0) {
            return RiskCheckResult(false, "Invalid stop loss")
        }

        val normalizedSide = normalizeSide(side)

        if (normalizedSide != OrderSide.BUY.name &&
            normalizedSide != OrderSide.SELL.name
        ) {
            return RiskCheckResult(false, "Invalid order side")
        }

        if (normalizedSide == OrderSide.BUY.name &&
            stopLoss >= entryPrice
        ) {
            return RiskCheckResult(
                false, "BUY stop loss must be below entry price"
            )
        }

        if (normalizedSide == OrderSide.SELL.name &&
            stopLoss <= entryPrice
        ) {
            return RiskCheckResult(
                false, "SELL stop loss must be above entry price"
            )
        }

        val activePositions = openPositions.filter { it.isOpen }
        val maxPositions = AppConfig.get().maxOpenPositions

        if (maxPositions > 0 && activePositions.size >= maxPositions) {
            return RiskCheckResult(
                false, "Maximum open positions limit reached"
            )
        }

        val requiredMargin = calculateRequiredMargin(
            normalizedSide, quantity, entryPrice
        )

        if (requiredMargin <= 0.0) {
            return RiskCheckResult(
                false, "Unable to calculate required margin"
            )
        }

        val usedMargin = calculateTotalUsedMargin(activePositions)
        val availableCapital = (capital - usedMargin).coerceAtLeast(0.0)

        if (requiredMargin > availableCapital) {
            return RiskCheckResult(
                allowed = false,
                message = "Insufficient available capital",
                requiredMargin = requiredMargin,
                availableCapital = availableCapital
            )
        }

        val estimatedRisk = abs(entryPrice - stopLoss) * quantity

        return RiskCheckResult(
            allowed = true,
            message = "Order passed risk checks",
            requiredMargin = requiredMargin,
            availableCapital = availableCapital,
            estimatedRisk = estimatedRisk
        )
    }

    fun calculatePositionMargin(position: Position): Double {
        if (!position.isOpen) return 0.0

        if (position.reservedMargin > 0.0 &&
            isValidNumber(position.reservedMargin)
        ) {
            return position.reservedMargin
        }

        return calculateRequiredMargin(
            position.side,
            position.quantity,
            position.entryPrice
        )
    }

    fun calculateTotalUsedMargin(
        positions: List<Position>
    ): Double {
        return positions
            .filter { it.isOpen }
            .sumByDouble { calculatePositionMargin(it) }
    }

    fun calculateAvailableCapital(
        totalCapital: Double,
        positions: List<Position>
    ): Double {
        if (!isValidNumber(totalCapital) || totalCapital <= 0.0) {
            return 0.0
        }

        val usedMargin = calculateTotalUsedMargin(positions)

        return (totalCapital - usedMargin).coerceAtLeast(0.0)
    }
}
